// Auth middleware — tenant-scoped policy gate
export async function authorize(request, response, next) {
  const token    = extractBearerToken(request.headers.authorization);
  const claims   = await verifyJwt(token, jwksClient);
  const tenantId = resolveTenant(request, claims);

  const decision = await policyEngine.evaluate({
    subject:  { userId: claims.sub, roles: claims.roles || [], tenantId },
    action:   request.method,
    resource: request.route.path,
  });

  if (!decision.allow) return response.status(403).json({ code: 'forbidden' });
  request.context = { claims, tenantId };
  next();
}
// Nightly listening digest — schedule this in a worker / cron
import { fetchMentions, aggregateMetrics, evaluateRules, buildDigestMarkdown } from "../src/lib/socialListening.js";

export async function nightlyDigest({ tenantId, uploader }) {
  const since   = Date.now() - 24 * 60 * 60_000;
  const posts   = await fetchMentions({ since });
  const metrics = aggregateMetrics(posts);
  const alerts  = evaluateRules({ metrics, baseline: null });
  const md      = buildDigestMarkdown({ window: "24h", metrics, alerts, posts });

  await uploader.put(`tenants/${tenantId}/digests/${new Date().toISOString()}.md`, md);
  if (alerts.length) await notify(tenantId, alerts);
  return { tenantId, mentions: metrics.total, alerts: alerts.length };
}

OPINIONATED VARIANTSCHOOSE A SHAPE

Three opinionated patterns extend the base template. Pick by team size, scaling profile, and how async the workload is. Full doc: ARCHITECTURE_VARIANTS.md.

VARIANT 1 · MODULAR MONOLITH
Best fit · early-to-mid SaaS, small platform team, strong domain cohesion.
  • One backend, strict internal modules
  • One Aurora DB, schema-separated
  • One worker (SQS + EventBridge)
  • Strength: fast delivery, simple ops
  • Tradeoff: lower independent scaling
VARIANT 2 · MICROSERVICES
Best fit · large surface, multiple teams, unequal scaling profiles.
  • Per-domain services + data ownership
  • API Gateway / ALB ingress
  • EventBridge + SQS event backbone
  • Strength: team autonomy + indep. deploy
  • Tradeoff: platform complexity
VARIANT 3 · EVENT-DRIVEN SAAS
Best fit · workflow-heavy, async integrations, analytics-rich products.
  • Command API + event backbone
  • Outbox / replay / dedupe components
  • Tenant-scoped events + observability
  • Strength: loose coupling, resilient
  • Tradeoff: eventual consistency complexity
APPLIED TO LITCENTRAL + JAMNET
Today the platform is best modeled as a modular monolith with an event-driven reporting + listening pipeline: one web app, one API surface, one database, and a listening / digest engine (src/lib/socialListening.js) that is already event-loop-shaped via startAutomation. When listening volume contends a single worker pool, lift just the listening + reporting subsystem into a separate service before splitting the rest.
